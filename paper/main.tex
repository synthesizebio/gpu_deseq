\begin{abstract}
\note{DRAFT.} Differential expression (DE) analysis with DESeq2 is a de-facto
standard for bulk RNA-seq, but its per-gene dispersion and generalized-linear-model
fits run sequentially on a single CPU core and dominate wall-clock time on modern,
many-sample datasets. We present \gpudeseq, a from-scratch reimplementation of the
DESeq2 pipeline in PyTorch that reproduces R DESeq2 1.30.1 to floating-point
precision at every pipeline step, while batching the embarrassingly-parallel
per-gene work across the GPU. We contribute (i) a bit-exact batched port of
DESeq2's Cox--Reid dispersion fitter; (ii) a CUDA-graph execution mode that
removes per-iteration kernel-launch overhead, enabled by a capture-safe
matrix-free linear solve; (iii) a fully fused Triton kernel that runs each gene's
entire Newton--Raphson dispersion loop in registers with per-gene early exit; and
(iv) a Levenberg--Marquardt reformulation of the apeGLM shrinkage optimizer that
fixes a convergence failure on extreme-fold-change genes. Across dataset sizes,
\gpudeseq{} matches R's differential-expression output (\fact{92/92 parity tests})
and is \fact{19--38$\times$} faster end-to-end than the fastest R configuration on
a single A100. \todo{finalize headline numbers + real-data validation sentence.}
\end{abstract}

\section{Background: the DESeq2 pipeline}
\note{Just enough DESeq2 to make the methods legible. Five steps, the per-gene
compute, and where R spends time.}
\begin{enumerate}[leftmargin=1.6em]
  \item \textbf{Normalization} --- median-of-ratios size factors.
  \item \textbf{Dispersion estimation} --- gene-wise CR-MLE, parametric trend,
        MAP shrinkage, outlier rule. \emph{The dominant cost.}
  \item \textbf{GLM fit / testing} --- IRLS negative-binomial GLM, Wald / LRT.
  \item \textbf{Significance} --- Cook's-distance filter, independent filtering,
        BH adjustment.
  \item \textbf{LFC shrinkage} --- apeGLM (Cauchy-prior MAP).
\end{enumerate}

sequentially in R's C++ core --- exactly the work a GPU batches.}

\subsection{Three execution modes}
\note{eager (per-op kernels), CUDA graph, Triton --- same math, different dispatch.
Table of tradeoffs (bit-identity, generality, effort).}

\subsection{CUDA-graph chunked replay}
\note{Problem: the NR loop fires $\sim$6k tiny kernels; the GPU is launch-bound.
A fixed full-length capture loses (early-exit); chunked replay of a $K$-iteration
graph with between-chunk convergence checks wins. cuSOLVER \texttt{slogdet}/\texttt{solve}
cannot be captured $\Rightarrow$ replaced by a capture-safe no-pivot LU
(\texttt{\_logdet\_and\_tr\_binv\_db}), matching \texttt{linalg} to $\sim$1e-15.}
\todo{Algorithm box for chunked replay + the LU substitution.}

\subsection{Triton full-loop fusion}
\note{One program per gene; counts/$\mu$/design loaded once; all iterations,
Cox--Reid $P\times P$ solve (unrolled LU, $P\in\{2,4\}$), NB log-likelihood and
gradient (\texttt{lgamma}/\texttt{digamma}/\texttt{log1p}), and the Armijo loop
run in registers; per-gene \texttt{while} early exit. digamma hand-implemented to
match ATen's \texttt{calc\_digamma}.}
\todo{Kernel structure figure; note $\sim$1e-14 vs R (not bit-identical to eager).}

\subsection{Fixing apeGLM shrinkage}
\note{The most novel piece. Diagnosis: the apeGLM Hessian is indefinite for
extreme-LFC genes ($\hat\beta\to\pm\infty$), so Newton ascends and overshoots to
the $\beta$ boundary; $\sim$27\% of genes never converge, burning all iterations
then a slow per-gene scipy rescue. Fix: per-gene Levenberg--Marquardt damping
(shrink $\lambda$ on success toward Newton, grow toward gradient descent) +
pinned-boundary convergence.}

\subsection{Scaling}
\note{Gene axis (2k vs 20k) and sample axis (2v2 pilot to 2000 samples). Key
findings: accelerators help most in the launch-bound small-sample regime
(Triton up to 7$\times$ over eager at n=6); at $>$1000 samples the loop becomes
compute-bound. Multi-core R only beats single-thread at $\geq$20k genes (fork
overhead below that).}

\subsection{Ablations}
\begin{itemize}[leftmargin=1.4em]
  \item Fixed-length vs chunked CUDA graph (0.8$\times$ $\to$ 5.5$\times$).
  \item Fixed-iteration vs per-gene early-exit Triton (fixes the 20k regression).
  \item apeGLM: Newton+scipy vs LM+pinned (58$\times$ on the step; parity kept).
  \item \texttt{GRAPH\_CHUNK} sweep (chunk 10 near-optimal).
\end{itemize}

\section{Limitations}
\note{Honest scope from README/RESULTS.}
\begin{itemize}[leftmargin=1.4em]
  \item Tested designs: single factor, additive multi-factor (P=4), continuous.
        Not yet: interaction designs, $>$3-level factors, $n<6$.
  \item Not implemented: \texttt{replaceOutliers} refit, \texttt{lfcShrink}
        types \texttt{normal}/\texttt{ashr}, \texttt{glmGamPoi}, VST/rlog.
  \item Triton path: $P\in\{2,4\}$ only (falls back to eager otherwise); not
        bit-identical to eager ($\sim$1e-14 vs R).
  \item Benchmarks: single A100, 12-core R host.
  \item DESeq2 versions other than 1.30.1 untested.
\end{itemize}
